        \begin{itemize}

          \item Tối ưu hóa xe nhàn rỗi của chủ xe cá nhân/doanh nghiệp.

        \end{itemize}

        \end{itemize}
      \end{block}
    \end{column}

      \begin{alertblock}{Thách thức SQA đối với Mioto.vn}
        \begin{itemize}

          \item \textbf{KYC nghiêm ngặt:} Xác minh Giấy phép lái xe (GPLX B1/B2/C), CCCD gắn chip.
          \item \textbf{Giao dịch tài chính thời gian thực:} Tích hợp đa cổng thanh toán (Alepay, MoMo, VNPay), chống Double Booking.
          \item \textbf{Ràng buộc pháp lý \& bảo hiểm:} Bảo hiểm vật chất chuyến đi (MIC, VNI), chính sách bồi cọc khi hủy.
        \end{itemize}
      \end{alertblock}
    \end{column}
  \end{columns}

\begin{frame}{1.2. Mục tiêu Nghiên cứu \& Phạm vi Kiểm thử}
  \begin{block}{Mục tiêu Đề tài (Objectives)}
    \begin{enumerate}
      \item Khảo sát, cào dữ liệu và phân tích kiến trúc vi dịch vụ thực tế của nền tảng Mioto.vn.
      \item Thiết lập \textbf{Kế hoạch kiểm thử (Test Plan)} chuẩn hóa theo tiêu chuẩn quốc tế \textbf{IEEE 829}.

      \item Ứng dụng kỹ thuật \textbf{Kiểm thử hộp trắng (White-box)}: Vẽ sơ đồ dòng điều khiển (CFG), tính độ phức tạp $V(G)$ và bao phủ 100\% Basis Paths thuật toán tính cước.

      \item Xây dựng kịch bản \textbf{Kiểm thử tự động (Automation Suite)} cho API (Postman) và UI E2E (Selenium Python).
    \end{enumerate}
  \end{block}

    \begin{itemize}
      \item \textbf{Số điện thoại (SĐT):}
        \begin{itemize}
          \item \textit{Valid Partition:} SĐT Việt Nam chuẩn 10 chữ số (03x, 05x, 07x, 08x, 09x), VD: \texttt{0912345678}.
          \item \textit{Boundary BVA:} 9 chữ số (thiếu số $\rightarrow$ Invalid); 11 chữ số (thừa số $\rightarrow$ Invalid); Rỗng.
        \end{itemize}
      \item \textbf{Mã xác thực OTP:}
        \begin{itemize}
          \item Đúng 6 chữ số trong 60 giây (Valid); Quá 60 giây (Invalid); Nhập sai liên tiếp 5 lần (Khóa 15p).
        \end{itemize}
      \item \textbf{Khoảng cách giao xe tận nơi ($d$):}
        \begin{itemize}
          \item $d \le 5.0\text{ km}$: Miễn phí giao xe (Biên trên miễn phí).

          \item Tối thiểu 24 giờ (1 ngày); Thời gian nhận xe phải sau thời điểm hiện tại ít nhất 2 giờ.

        $$V(G) = P + 1 = 6 + 1 = \mathbf{7}$$
        \textbf{Phương pháp 3 (Vùng khép kín $R$):}
        $$V(G) = \text{Enclosed Regions} + 1 = 6 + 1 = \mathbf{7}$$
      \end{block}
      \begin{alertblock}{Kết luận Hộp trắng}
        Có \textbf{chính xác 7 đường đi cơ sở độc lập (Basis Paths)} cần bao phủ.
